cmake_minimum_required(VERSION 3.8)
project(maicon_arena_gz)
find_package(ament_cmake REQUIRED)
set(MAICON_GAZEBO_SIMULATOR "fortress" CACHE STRING "Gazebo library family: fortress or harmonic")
add_library(maicon_opaque_material_system SHARED src/opaque_material_system.cc)
target_compile_features(maicon_opaque_material_system PRIVATE cxx_std_17)
find_package(Qt5 REQUIRED COMPONENTS Quick)
add_library(MaiconOpaqueMaterialGui SHARED src/opaque_material_gui.cc src/opaque_material_gui.qrc)
target_compile_features(MaiconOpaqueMaterialGui PRIVATE cxx_std_17)
set_target_properties(MaiconOpaqueMaterialGui PROPERTIES AUTOMOC ON AUTORCC ON)
if(MAICON_GAZEBO_SIMULATOR STREQUAL "fortress")
  find_package(ignition-gazebo6 REQUIRED)
  find_package(ignition-rendering6 REQUIRED)
  find_package(ignition-plugin1 REQUIRED COMPONENTS register)
  find_package(ignition-gui6 REQUIRED)
  target_link_libraries(maicon_opaque_material_system PRIVATE
    ignition-gazebo6::ignition-gazebo6 ignition-rendering6::ignition-rendering6 ignition-plugin1::register)
  target_link_libraries(MaiconOpaqueMaterialGui PRIVATE
    ignition-gui6::ignition-gui6 ignition-rendering6::ignition-rendering6 ignition-plugin1::register Qt5::Quick)
elseif(MAICON_GAZEBO_SIMULATOR STREQUAL "harmonic")
  find_package(gz-sim8 REQUIRED)
  find_package(gz-rendering8 REQUIRED)
  find_package(gz-plugin2 REQUIRED COMPONENTS register)
  find_package(gz-gui8 REQUIRED)
  target_compile_definitions(maicon_opaque_material_system PRIVATE MAICON_GZ_HARMONIC)
  target_compile_definitions(MaiconOpaqueMaterialGui PRIVATE MAICON_GZ_HARMONIC)
  target_link_libraries(maicon_opaque_material_system PRIVATE
    gz-sim8::gz-sim8 gz-rendering8::gz-rendering8 gz-plugin2::register)
  target_link_libraries(MaiconOpaqueMaterialGui PRIVATE
    gz-gui8::gz-gui8 gz-rendering8::gz-rendering8 gz-plugin2::register Qt5::Quick)
else()
  message(FATAL_ERROR "MAICON_GAZEBO_SIMULATOR must be fortress or harmonic")
endif()
install(TARGETS maicon_opaque_material_system LIBRARY DESTINATION lib)
install(TARGETS MaiconOpaqueMaterialGui LIBRARY DESTINATION lib)
install(DIRECTORY config launch media models worlds docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY reference DESTINATION share/${PROJECT_NAME})
install(DIRECTORY validation/ DESTINATION share/${PROJECT_NAME}/validation
  FILES_MATCHING PATTERN "*.png" PATTERN "*.jpg" PATTERN "*.jpeg" PATTERN "*.json")
install(DIRECTORY scripts/ DESTINATION share/${PROJECT_NAME}/scripts FILES_MATCHING PATTERN "*.py")
file(GLOB arena_scripts "${CMAKE_CURRENT_SOURCE_DIR}/scripts/*.py")
install(PROGRAMS ${arena_scripts} DESTINATION lib/${PROJECT_NAME})
ament_environment_hooks(hooks/arena_resources.dsv.in)
ament_package()
